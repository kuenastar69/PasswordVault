Pseudocode Linked List
// Struktur Node class Node { service : String username : String password : String // tersimpan dalam bentuk terenkripsi category : String next : Node | null } // Insert — O(1), sisip di HEAD insert(service, user, pass, cat): newNode = new Node(service, user, encrypt(pass), cat) newNode.next = head head = newNode // Delete — O(n), cari lalu hapus node delete(id): prev = null, cur = head while cur != null: if cur.id == id: if prev == null: head = cur.next else: prev.next = cur.next return cur prev = cur; cur = cur.next

// Struktur Stack — prinsip LIFO class Stack { items : Array } // Push — O(1), tambah ke top push(action): items.append(action) // Pop — O(1), ambil dari top (LIFO) pop(): if items.isEmpty(): return null return items.removeLast() // Enkripsi Caesar Cipher ROT13 encrypt(text): for each char in text: if isAlpha(char): shift char by 13 positions return encryptedText
